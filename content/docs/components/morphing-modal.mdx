## Usage

```tsx
import {
  MorphingModal,
  MorphingModalContent,
  MorphingModalDescription,
  MorphingModalTitle,
  MorphingModalTrigger,
  MorphingModalView,
} from "@/components/transitions/morphing-modal";
import { BACK, FORWARD } from "@/components/transitions/ui-transition";
```

```tsx
const [view, setView] = useState("menu");

const go = (next: string) =>
  startTransition(() => {
    addTransitionType(next === "menu" ? BACK : FORWARD);
    setView(next);
  });

<MorphingModal>
  <MorphingModalTrigger>Share</MorphingModalTrigger>
  <MorphingModalContent aria-label="Share project">
    {view === "menu" ? (
      <MorphingModalView id="menu">
        <MorphingModalTitle>Share project</MorphingModalTitle>
        <button onClick={() => go("link")}>Copy link</button>
      </MorphingModalView>
    ) : (
      <MorphingModalView id="link">
        <button onClick={() => go("menu")}>Back</button>
        <MorphingModalDescription>
          https://example.com/p/1
        </MorphingModalDescription>
      </MorphingModalView>
    )}
  </MorphingModalContent>
</MorphingModal>;
```

The modal is a native `<dialog>`: focus stays inside it, Escape and a click on the backdrop close it, and focus returns to the trigger.

## API Reference

### MorphingModal

| Prop           | Type                      | Default | Description                            |
| -------------- | ------------------------- | ------- | -------------------------------------- |
| `open`         | `boolean`                 | -       | Controlled open state.                 |
| `defaultOpen`  | `boolean`                 | `false` | Initial open state when uncontrolled.  |
| `onOpenChange` | `(open: boolean) => void` | -       | Called when the modal opens or closes. |

### MorphingModalContent

| Prop         | Type      | Default | Description                                 |
| ------------ | --------- | ------- | ------------------------------------------- |
| `showClose`  | `boolean` | `true`  | Renders the close button in the top corner. |
| `aria-label` | `string`  | -       | Accessible name, or use `aria-labelledby`.  |

### MorphingModalView

| Prop | Type     | Default | Description                                                   |
| ---- | -------- | ------- | ------------------------------------------------------------- |
| `id` | `string` | -       | Identifies the view; a new `id` slides the views and resizes. |

`useMorphingModal()` returns `{ open, setOpen }` from inside the modal.
