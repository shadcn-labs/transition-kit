## Usage

```tsx
import {
  AnimatedList,
  AnimatedListItem,
  useAnimatedList,
} from "@/components/transitions/animated-list";
```

```tsx
const { items, add, remove, move, sort, set } = useAnimatedList([
  { id: 1, title: "Deploy finished" },
  { id: 2, title: "Payment received" },
]);

<AnimatedList aria-label="Notifications">
  {items.map((item) => (
    <AnimatedListItem key={item.id} id={item.id}>
      {item.title}
      <button type="button" onClick={() => remove(item.id)}>
        Dismiss
      </button>
    </AnimatedListItem>
  ))}
</AnimatedList>;
```

```tsx
add({ id: 3, title: "New comment" }, 0); // insert at index 0 (default: end)
remove(2); // by id
move(1, 0); // move id 1 to index 0
sort((a, b) => a.title.localeCompare(b.title));
set((items) => items.filter((item) => !item.read)); // any update
```

```tsx
startTransition(() => setItems((items) => items.slice(1)));
```

## API Reference

### AnimatedListItem

| Prop | Type               | Default | Description                                                                         |
| ---- | ------------------ | ------- | ----------------------------------------------------------------------------------- |
| `id` | `string \| number` | -       | Unique identity for the row, stable between renders. Render it with the same `key`. |

### useAnimatedList

| Return   | Type                                            | Description                                          |
| -------- | ----------------------------------------------- | ---------------------------------------------------- |
| `items`  | `T[]`                                           | Current list contents.                               |
| `add`    | `(item: T, index?: number) => void`             | Inserts an item at the index (default: end).         |
| `remove` | `(id: string \| number) => void`                | Removes the item with the given id.                  |
| `move`   | `(id: string \| number, index: number) => void` | Moves the item with the given id to an index.        |
| `sort`   | `(compare: (a: T, b: T) => number) => void`     | Reorders items with a compare function.              |
| `set`    | `(items: T[] \| ((items: T[]) => T[])) => void` | Replaces items with an array or an updater function. |

```ts
const { items, add, remove, move, sort, set } = useAnimatedList(initial, {
  getId,
});
```
