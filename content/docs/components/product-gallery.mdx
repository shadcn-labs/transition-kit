## Usage

```tsx
import {
  ProductGallery,
  ProductGalleryMain,
  ProductGalleryThumbnails,
} from "@/components/transitions/product-gallery";
```

```tsx
const images = [
  { id: "front", src: "/shoe-front.jpg", alt: "Front view" },
  { id: "side", src: "/shoe-side.jpg", alt: "Side view" },
  { id: "sole", src: "/shoe-sole.jpg", alt: "Sole detail" },
];

<ProductGallery images={images} className="max-w-sm">
  <ProductGalleryMain />
  <ProductGalleryThumbnails />
</ProductGallery>;
```

```tsx
const [index, setIndex] = useState(0);

<ProductGallery images={images} index={index} onIndexChange={setIndex}>
  <ProductGalleryMain />
  <ProductGalleryThumbnails />
</ProductGallery>;
```

## API Reference

### ProductGallery

| Prop            | Type                      | Default | Description                                               |
| --------------- | ------------------------- | ------- | --------------------------------------------------------- |
| `images`        | `ProductGalleryImage[]`   | -       | Images available in the gallery.                          |
| `index`         | `number`                  | -       | Controlled selected image index.                          |
| `defaultIndex`  | `number`                  | `0`     | Initial selected index when uncontrolled.                 |
| `onIndexChange` | `(index: number) => void` | -       | Called with the target index after the new image decodes. |

### ProductGalleryImage

| Field  | Type              | Default | Description                                                               |
| ------ | ----------------- | ------- | ------------------------------------------------------------------------- |
| `id`   | `string`          | -       | Stable identity that pairs the hero and thumbnail images.                 |
| `alt`  | `string`          | -       | Image label used for the thumbnail name and the live-region announcement. |
| `src`  | `string`          | -       | Image URL rendered as an `<img>` with `object-fit: cover`.                |
| `node` | `React.ReactNode` | -       | Custom content rendered instead of `src`; it should fill its box.         |

### ProductGalleryMain

| Prop       | Type      | Default | Description                                 |
| ---------- | --------- | ------- | ------------------------------------------- |
| `controls` | `boolean` | `true`  | Shows previous/next buttons over the image. |
