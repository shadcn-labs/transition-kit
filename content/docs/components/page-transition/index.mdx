```tsx
"use client";

import { useState } from "react";

import {
  navigateWithTransition,
  PageTransition,
} from "@/components/transitions/page-transition";

const TABS = ["overview", "activity", "settings"];

export const Tabs = () => {
  const [tab, setTab] = useState("overview");

  return (
    <>
      <nav>
        {TABS.map((next) => (
          <button
            key={next}
            type="button"
            onClick={() =>
              navigateWithTransition(() => setTab(next), {
                direction:
                  TABS.indexOf(next) < TABS.indexOf(tab) ? "back" : "forward",
              })
            }
          >
            {next}
          </button>
        ))}
      </nav>
      <PageTransition id={tab} transition="slide" className="bg-background">
        <TabPanel tab={tab} />
      </PageTransition>
    </>
  );
};
```

## API Reference

### PageTransition

| Prop         | Type                    | Default  | Description                                                                                                                       |
| ------------ | ----------------------- | -------- | --------------------------------------------------------------------------------------------------------------------------------- |
| `id`         | `string`                | -        | Identifies the current page, usually the pathname. The old page exits and the new one enters when it changes inside a Transition. |
| `transition` | `string`                | `"fade"` | Style slug. Fade ships with the core.                                                                                             |
| `duration`   | `number`                | -        | Milliseconds. Overrides the style's timing.                                                                                       |
| `easing`     | `string`                | -        | Overrides the style's easing.                                                                                                     |
| `children`   | `ReactNode`             | -        | The routed content.                                                                                                               |
| `...props`   | `ComponentProps<"div">` | -        | Spread onto the `<div data-slot="page-transition">` that wraps each page and is snapshotted.                                      |

### navigateWithTransition

```ts
navigateWithTransition(navigate: () => void, options?: { direction?: "forward" | "back" }): void
```

| Param               | Type                  | Default     | Description                                                                                                                          |
| ------------------- | --------------------- | ----------- | ------------------------------------------------------------------------------------------------------------------------------------ |
| `navigate`          | `() => void`          | -           | Anything updating the state `id` comes from: `router.push`, React Router's `navigate`, or `setState`. Runs inside `startTransition`. |
| `options.direction` | `"forward" \| "back"` | `"forward"` | `"back"` plays directional styles in reverse.                                                                                        |

### PAGE_BACK_TYPE

| Field            | Type          | Description                                                                            |
| ---------------- | ------------- | -------------------------------------------------------------------------------------- |
| `PAGE_BACK_TYPE` | `"page-back"` | Pass it to your router's own transition types API instead of `navigateWithTransition`. |
